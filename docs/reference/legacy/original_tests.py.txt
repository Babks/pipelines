import math
import sys
import unittest
from pathlib import Path

ROOT = Path(__file__).resolve().parents[1]
sys.path.insert(0, str(ROOT))

from vtd_corrosion_mvp import calculate_single_defect, run_calculation


SAMPLE = Path('/mnt/data/0936a617-4998-43d9-8020-97b5606c3f69.xlsx')


class FormulaTests(unittest.TestCase):
    def test_sto_example_v1(self):
        r = calculate_single_defect(
            1420, 18.7, 588.4, 500, 5.3, 7.4, 1.1, 1.34, 1.1, 0.9
        )
        self.assertAlmostEqual(r['Q'], 1.979541, places=5)
        self.assertAlmostEqual(r['p_cr_mpa'], 13.13363, places=4)
        self.assertTrue(r['passes'])

    def test_sto_example_v2(self):
        r = calculate_single_defect(
            1420, 18.7, 588.4, 1000, 6.8, 7.4, 1.1, 1.34, 1.1, 0.9
        )
        self.assertAlmostEqual(r['Q'], 3.560102, places=5)
        self.assertAlmostEqual(r['p_cr_mpa'], 11.13040, places=4)
        self.assertFalse(r['passes'])


class IntegrationTests(unittest.TestCase):
    @unittest.skipUnless(SAMPLE.exists(), 'sample xlsx is not mounted')
    def test_defect_91199328_from_xlsx(self):
        r = run_calculation(SAMPLE, defect_id='91199328')
        self.assertEqual(r['defect']['n_tube'], 1364)
        self.assertEqual(r['isolation']['status'], 'single')
        calc = r['calculation']
        self.assertIsNotNone(calc)
        self.assertAlmostEqual(calc['p_cr_mpa'], 13.595063, places=5)
        self.assertAlmostEqual(calc['p_allow_mpa'], 9.385703, places=5)
        self.assertTrue(calc['passes'])
        self.assertAlmostEqual(calc['pf_db_mpa'], 13.596, places=3)
        self.assertLess(abs(calc['pf_delta_mpa']), 0.002)


if __name__ == '__main__':
    unittest.main()
